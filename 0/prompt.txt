fn snapshot_verified_entry(
    artifact_root: &Path,
    slot: &ReleaseArtifactSlot,
    expected: &ValidatedArtifactEntry,
) -> Result<ArtifactSnapshot, GraphError> {
    let path = resolve_asset_path(artifact_root, slot.file_name())?;
    if path != expected.path {
        return Err(GraphError::UnsafeArtifactPath {
            path,
            reason: "artifact path changed after validation",
        });
    }
    let bytes = fs::read(&path).map_err(|error| GraphError::ArtifactRead {
        path: path.clone(),
        message: error.to_string(),
    })?;
    if ArtifactDigest::from_bytes(&bytes) != expected.digest || bytes.len() as u64 != expected.byte_length {
        return Err(GraphError::ArtifactBytesMismatch { path });
    }
    Ok(ArtifactSnapshot {
        slot: slot.clone(),
        bytes,
    })
}
