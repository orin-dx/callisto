callisto_parse_version() {
  local requested="$1"
  case "$requested" in
    latest)
      CALLISTO_REQUEST_KIND="latest"
      CALLISTO_REQUESTED_VERSION=""
      ;;
    callisto@*)
      CALLISTO_REQUEST_KIND="exact"
      CALLISTO_REQUESTED_VERSION="${requested#callisto@}"
      ;;
    *)
      CALLISTO_REQUEST_KIND="exact"
      CALLISTO_REQUESTED_VERSION="$requested"
      ;;
  esac

  if [[ "$CALLISTO_REQUEST_KIND" == "exact" \
    && ! "$CALLISTO_REQUESTED_VERSION" =~ ^[0-9]+\.[0-9]+\.[0-9]+$ ]]; then
