fn observe_tag(
    artifacts: &ValidatedArtifactManifest<'_, '_>,
    validated: &ValidatedReleaseIntent<'_>,
    release: &ProductReleaseId,
    name: &TagName,
    target: &callisto_model::CommitSha,
    annotation: &str,
    remote_name: &str,
) -> OperationObservation {
    let expected_annotation = TagAnnotationDigest::from_annotation(annotation);
    match GitAccess::discover(validated.root(), validated.runner()).observe_remote_annotated_tag(
        remote_name,
        name,
        target,
        &expected_annotation,
    ) {
        Ok(RemoteAnnotatedTagObservation::Absent) => OperationObservation::Absent,
        Ok(RemoteAnnotatedTagObservation::Conflict(conflict)) => {
            OperationObservation::Conflict(ExternalConflict::RemoteTag(conflict))
        }
        Ok(RemoteAnnotatedTagObservation::Exact) => {
            let expectations = artifacts.expectations();
            OperationObservation::Exact(ReleaseObservationEvidence::remote_tag(
                release.clone(),
                name.as_str(),
                target.clone(),
                expectations.tag_annotation_digest().clone(),
            ))
        }
        Err(_) => OperationObservation::Unverifiable(ObservationFailure::Command {
            provider: ObservationProvider::Git,
        }),
    }
}

pub fn reconcile_release_execution(
    intent: &ReleaseIntent,
    state: &ReleaseExecutionState,
) -> Result<ReconciledReleaseExecution, GraphError> {
    state
        .validate_for_intent(intent)
        .map_err(|source| GraphError::ReleaseExecutionState { source })?;
    let prerequisites: BTreeMap<_, _> = intent
        .operations()
        .iter()
        .map(|operation| (operation.id().clone(), operation.prerequisites().to_vec()))
        .collect();
    let mut eligible = Vec::new();
    for operation in intent.operations() {
        if state.operation_state(operation.id()) != Some(ReleaseOperationState::Pending) {
            continue;
        }
        let mut visited = BTreeSet::new();
        if prerequisites_satisfied_transitively(operation.id(), &prerequisites, state, &mut visited) {
            eligible.push(operation.id().clone());
        }
    }
    Ok(ReconciledReleaseExecution { eligible })
}

fn prerequisites_satisfied_transitively(
    id: &ReleaseOperationId,
    prerequisites: &BTreeMap<ReleaseOperationId, Vec<ReleaseOperationId>>,
    state: &ReleaseExecutionState,
    visited: &mut BTreeSet<ReleaseOperationId>,
) -> bool {
    if !visited.insert(id.clone()) {
        return false;
    }
    let result = prerequisites.get(id).is_some_and(|direct| {
        direct.iter().all(|prerequisite| {
            matches!(
                state.operation_state(prerequisite),
                Some(ReleaseOperationState::Published | ReleaseOperationState::AlreadySatisfied)
            ) && prerequisites_satisfied_transitively(prerequisite, prerequisites, state, visited)
        })
    });
    visited.remove(id);
    result
}
