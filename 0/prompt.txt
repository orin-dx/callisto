/// Builds the one canonical npm invocation whose only package input is a
/// private, already-validated tarball. Workspace package-manager selection is
/// intentionally irrelevant: pnpm, Yarn Berry, and Bun do not share one
/// stable exact-tarball publish contract, while the registry protocol is npm's
/// and the canonical npm client accepts the artifact directly.
pub(super) fn npm_tarball_publish_argv(
    tarball: &Path,
    tag: &str,
    access: NpmAccess,
    registry: &str,
) -> Argv {
    let tarball_argument = tarball.to_string_lossy().into_owned();
    let mut args = vec!["publish".to_string(), tarball_argument, "--ignore-scripts".to_string()];
    args.extend([
        "--tag".to_string(),
        tag.to_string(),
        "--access".to_string(),
        match access {
            NpmAccess::Public => "public".to_string(),
            NpmAccess::Restricted => "restricted".to_string(),
        },
        "--registry".to_string(),
        registry.to_string(),
    ]);
    Argv {
        program: "npm".to_string(),
        args,
    }
}
