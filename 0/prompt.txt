    package: &ReleasePackageId,
    version: &Version,
    configured: Option<&[crate::config::ReleaseArtifactDescriptor]>,
) -> Result<Vec<(RegistryArtifactFormat, String)>, GraphError> {
    match package.ecosystem() {
        Ecosystem::Cargo => Ok(vec![(
            RegistryArtifactFormat::CargoCrate,
            format!("{}-{}.crate", package.name(), version.render()),
        )]),
        Ecosystem::Npm => Ok(vec![(
            RegistryArtifactFormat::NpmTarball,
            format!(
                "{}-{}.tgz",
                package.name().trim_start_matches('@').replace('/', "-"),
                version.render()
            ),
        )]),
        Ecosystem::Pypi => configured
            .filter(|descriptors| !descriptors.is_empty())
            .and_then(|descriptors| {
                descriptors
                    .iter()
                    .map(|descriptor| {
                        descriptor
                            .file_name_for(version)
                            .map(|file_name| (descriptor.format(), file_name))
                    })
                    .collect::<Option<Vec<_>>>()
            })
            .ok_or_else(|| GraphError::ReleaseInvariant {

                ),
            }),
        _ => Err(GraphError::UnsupportedRelease {

        }),
    }
