/// Invocation-local provider cache. A convergence pass shares one session so
/// every GitHub operation sees one release/asset snapshot and every PyPI
/// operation for a version sees one distribution roster.
#[derive(Default)]
pub(super) struct ObservationSession {
    github_releases: BTreeMap<(String, String), CachedObservation<GitHubRelease>>,
    pypi_versions: BTreeMap<(String, String, String), CachedObservation<BTreeMap<String, PypiDistribution>>>,
}

impl ObservationSession {
    /// Starts a fresh observation phase after any external effect.
    ///
    /// Provider snapshots are invocation-local, but a dispatcher can observe,
    /// mutate, and then verify within one invocation. Clearing every cached
    /// provider fact is deliberately broader than operation-specific
    /// invalidation: one effect can change both a release roster and the state
    /// observed by a dependent operation.
    pub(super) fn invalidate_after_effect(&mut self) {
        self.github_releases.clear();
        self.pypi_versions.clear();
    }
}

fn validated_registry_key(raw: String) -> Result<RegistryKey, successor::SuccessorError> {
    if raw.is_empty()
        || raw.len() > 128
        || !raw
            .bytes()
            .all(|byte| byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'_'))
    {
        return Err(successor::SuccessorError::MalformedRegistryKey { raw });
    }
    Ok(RegistryKey(raw))
}
