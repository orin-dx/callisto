     pub fn compare(&self, other: &Version) -> Result<std::cmp::Ordering, GrammarMismatch> {
         if self.grammar != other.grammar {
             return Err(GrammarMismatch { left: self.grammar, right: other.grammar });
         }
         match (&self.parsed, &other.parsed) {
             (ParsedVersion::SemVer(a), ParsedVersion::SemVer(b)) => {
                 Ok((a.major, a.minor, a.patch, &a.pre).cmp(&(b.major, b.minor, b.patch, &b.pre)))
             }
         }
     }
