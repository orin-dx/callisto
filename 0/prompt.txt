    latest)

    callisto@*)
      CALLISTO_REQUEST_KIND="exact"
      CALLISTO_REQUESTED_VERSION="${requested#callisto@}"
      ;;

      CALLISTO_REQUESTED_VERSION="$requested"

  if [[ "$CALLISTO_REQUEST_KIND" == "exact" \
