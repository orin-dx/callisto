/// child can never deadlock waiting on a full OS pipe buffer -- this

const MAX_CAPTURED_OUTPUT_BYTES: usize = 10 * 1024 * 1024;

pub struct CliCommandRunner;

impl CommandRunner for CliCommandRunner {
    fn run(&self, program: &str, args: &[&str], cwd: &Path) -> Result<CommandOutput, CommandError> {
        let output = std::process::Command::new(program)
            .args(args)
            .current_dir(cwd)
            .stdin(Stdio::null())
            .stdout(Stdio::piped())
            .stderr(Stdio::piped())
            .output();

        match output {
            Ok(o) => {
                let stderr = String::from_utf8_lossy(&o.stderr).into_owned();
                // Print stderr to the terminal before constructing the output
                // so the caller sees it even if they don't inspect the field.
                if !stderr.is_empty() {
                    eprint!("{stderr}");
                }

                    exit_code: o.status.code(),
                    stdout: String::from_utf8_lossy(&o.stdout).into_owned(),

### High — local test command can hide CI failures

- **Status:** CONFIRMED
- **Location:** [justfile](/private/tmp/callisto-release-execution-foundation/justfile:15)
- **Root cause:** `just test` runs `cargo nextest … || moon run :test`. Any Nextest failure—including “Nextest is not installed,” a configuration error, or a real failing test—falls back to a different runner.
- **Why it matters:** CI runs [justfile](/private/tmp/callisto-release-execution-foundation/justfile:21) `just test-ci`, which has no fallback. This is exactly how we got “local tests passed” while CI immediately failed due to missing Nextest.
- **Recommendation:** Make `just test` fail if Nextest is installed but fails. Provide a separately named fallback command only for developers without Nextest, and make the pre-push/PR workflow use the same `test-ci` path locally.

### High — raw subprocess stderr can leak secrets before redaction

- **Status:** CONFIRMED
- **Location:** [runner.rs](/private/tmp/callisto-release-execution-foundation/crates/callisto-cli/src/runner.rs:30)
- **Root cause:** `CliCommandRunner::run` unconditionally writes raw child stderr to the terminal. Many callers redact that same stderr later when constructing diagnostics, but the secret has already reached CI logs.
- **Failing scenario:** A registry client or Git command writes a token-bearing URL or credential to stderr. The runner prints it; later redaction cannot undo the log leak.
- **Recommendation:** Apply the same live-output redaction policy already used by `run_with_timeout`, or make non-interactive/JSON execution quiet and return redacted diagnostics only. Add a re-exec regression test specifically for `run`, not only `run_with_timeout`.

### Medium — test fixture portability is still structurally fragile

- **Status:** CONFIRMED
- **Location:** [durable_release_e2e_tests.rs](/private/tmp/callisto-release-execution-foundation/crates/callisto-cli/tests/durable_release_e2e_tests.rs:188)
- **Root cause:** The test embeds shell programs as Rust strings, alters `PATH`, and currently delegates Git through an absolute `/usr/bin/git`.
- **Why it matters:** This is the same class of problem that produced the CI break. The current GitHub macOS/Linux matrix is good, but other Unix environments need not place Git there.
- **Recommendation:** Restore a validated real-Git resolver captured before modifying `PATH`; keep boundary doubles deliberately small; add a fixture self-test that validates its emitted GitHub API body as JSON and its HTTP framing before release assertions run.
