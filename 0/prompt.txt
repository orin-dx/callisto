  ```rust
  fn run_with_timeout(&self, program: &str, args: &[&str], cwd: &Path, timeout: Duration) -> Result<CommandOutput, CommandError> {
      let mut child = Command::new(program).args(args).current_dir(cwd)
          .stdin(Stdio::null()).stdout(Stdio::piped()).stderr(Stdio::piped()).spawn()?;
      
      let stdout_handle = child.stdout.take().unwrap();
      let stderr_handle = child.stderr.take().unwrap();
      let stdout_thread = thread::spawn(move || { let mut buf = String::new(); drop(BufReader::new(stdout_handle).read_to_string(&mut buf)); buf });
      let stderr_thread = thread::spawn(move || { let mut buf = String::new(); drop(BufReader::new(stderr_handle).read_to_string(&mut buf)); buf });
      
      let deadline = Instant::now() + timeout;
      let status = loop {
          match child.try_wait()? {
              Some(s) => break s,
              None => {
                  if Instant::now() >= deadline {
                      drop(child.kill()); drop(child.wait());
                      drop(stdout_thread.join()); drop(stderr_thread.join());
                      return Err(CommandError::TimedOut { program: program.to_string(), seconds: timeout.as_secs() });
                  }
                  thread::sleep(Duration::from_millis(50));
              }
          }
      };
      // collect stdout/stderr, return CommandOutput
  }
  
  // Tests:
  #[test]
  fn run_with_timeout_kills_slow_process_and_returns_timed_out() {
      let runner = CliCommandRunner;
      let err = runner.run_with_timeout("sleep", &["1"], Path::new("."), Duration::from_millis(100)).unwrap_err();
      assert!(matches!(err, CommandError::TimedOut { .. }));
  }
  #[test]
  fn run_with_timeout_returns_output_for_fast_process() {
      let runner = CliCommandRunner;
      let out = runner.run_with_timeout("true", &[], Path::new("."), Duration::from_secs(5)).unwrap();
      assert!(out.success());
  }
  ```
