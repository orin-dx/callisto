#[allow(clippy::too_many_arguments)]
pub fn reconcile_release_from_artifacts<L: ProjectLocator, R: CommandRunner>(
    root: &Path,
    locator: &L,
    runner: &R,
    explicit_state_directory: Option<&Path>,
    intent: ReleaseIntent,
    artifact_manifest: ReleaseArtifactManifest,
    artifact_directory: &Path,
) -> Result<ReleaseReadinessReport, GraphError> {
    let capability =
        validate_release_intent_with_state_directory(root, locator, runner, explicit_state_directory, intent)?;
    let artifacts = verify_artifact_manifest(&capability, artifact_manifest, artifact_directory, runner)?;
    let mut session = ObservationSession::default();
    let operations = capability
        .intent()
        .operations()
        .iter()
        .map(|operation| ReleaseReadinessOperation {
            operation: operation.id().clone(),
            observation: readiness_state(observe_prepared(&mut session, &artifacts, operation.id())),
        })
        .collect();
    Ok(ReleaseReadinessReport { operations })
}
