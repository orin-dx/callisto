    fn validate_slot_roster(
        release: &ProductReleaseId,
        destination: &ReleaseDestinationBinding,
        approved: &[ApprovedRegistryArtifact],
        slots: &[ReleaseArtifactSlot],
    ) -> Result<(), SuccessorError> {

            return Err(SuccessorError::DuplicateSlot);
        }
        for slot in slots {
            slot.validate(release)?;
        }
        let mut product_policies = slots.iter().filter_map(|slot| match slot {
            ReleaseArtifactSlot::ProductAsset { provenance_policy, .. } => Some(provenance_policy),
            _ => None,
        });
        if let Some(first) = product_policies.next() {
            if product_policies.any(|policy| policy != first) {

                    format,

                    file_name: file_name.clone(),
                }),
                _ => None,
            })
            .collect::<BTreeSet<_>>();
        if registry_slots.len()
