- Added `ShellGit::head_sha()` to `impl GitDataSource for ShellGit<'_>`:
```rust
fn head_sha(&self) -> Result<CommitSha, VcsError> {
    let output = self.runner.run("git", &["rev-parse", "HEAD"], &self.root)?;
    if !output.success() {
        return Err(VcsError::Git(format!(
            "`git rev-parse HEAD` failed in `{}`: {}", self.root.display(), output.stderr
        )));
    }
    let sha_str = output.stdout_trimmed();
    CommitSha::parse(sha_str).map_err(|e| VcsError::Git(format!(
        "could not parse HEAD SHA `{sha_str}`: {e}"
    )))
}
```
