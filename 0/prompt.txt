fn parse_http_stream(stream: &str) -> Option<(u16, &str)> {
    let (headers, body) = stream.rsplit_once("\r\n\r\n").or_else(|| stream.rsplit_once("\n\n"))?;
    let status = headers
        .lines()
        .rev()
        .find(|line| line.trim_end_matches('\r').starts_with("HTTP/"))?

        .nth(1)?
        .parse()
        .ok()?;
    Some((status, body))
}

fn cargo_metadata_url(endpoint: &str, package: &str, version: &Version) -> Option<String> {
    let mut url = url::Url::parse(endpoint).ok()?;
    if url.path_segments()?.next_back() != Some("new") {
        return None;
    }
    let rendered_version = version.render();
    {
        let mut segments = url.path_segments_mut().ok()?;
        segments.pop_if_empty();
        segments.pop();
        segments.extend([package, rendered_version]);
    }
    Some(url.to_string())
}

fn append_url_segments(base: &str, segments: &[&str]) -> Option<String> {
    let mut url = url::Url::parse(base).ok()?;
    {
        let mut path = url.path_segments_mut().ok()?;
        path.pop_if_empty();
        path.extend(segments.iter().copied());
    }
    Some(url.to_string())
}

fn verify_github_attestation<R: CommandRunner>(
    path: &Path,
    policy: &ProvenancePolicy,
    source_commit: &callisto_model::CommitSha,
    runner: &R,
) -> Result<(), GraphError> {
    let repository_slug = policy.repository().as_slug();
    let workflow = format!("{repository_slug}/{}", policy.workflow_path());
    let path_argument = path.to_string_lossy();
    let args = [
        "attestation",
        "verify",
        path_argument.as_ref(),
        "--repo",
        repository_slug.as_str(),
        "--signer-workflow",
        workflow.as_str(),
        "--signer-digest",
        policy.workflow_commit().as_str(),
        "--source-digest",
        source_commit.as_str(),
        "--deny-self-hosted-runners",
    ];
    let output = runner
        .run_with_timeout(
            "gh",
            &args,
            path.parent().unwrap_or_else(|| Path::new(".")),
            ATTESTATION_TIMEOUT,
        )
        .map_err(|error| GraphError::ArtifactAttestation {
            path: path.to_path_buf(),
            message: error.to_string(),
        })?;
    if output.success() {
        Ok(())
    } else {
        Err(GraphError::ArtifactAttestation {
            path: path.to_path_buf(),
            message: "GitHub attestation verification failed".to_owned(),
        })
    }
}
