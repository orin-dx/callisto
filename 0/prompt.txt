pub(crate) fn resolve_target_package<'a>(
    packages: impl Iterator<Item = &'a Package>,
    id: &PackageId,
) -> Result<Option<&'a Package>, GraphError> {
    id.resolve_unique(packages, |p| &p.id)

            candidates: candidates.iter().map(|p| p.id.clone()).collect(),
        })
}
