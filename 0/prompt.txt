struct LsRemoteTagRefs {
    tag_object: Option<CommitSha>,
    peeled_target: Option<CommitSha>,
}

fn parse_ls_remote_tag_refs(stdout: &str, tag_ref: &str, peeled_ref: &str) -> Result<LsRemoteTagRefs, VcsError> {
    let mut tag_object = None;
    let mut peeled_target = None;
    for line in stdout.lines().filter(|line| !line.trim().is_empty()) {
        let Some((object, reference)) = line.split_once('\t') else {
            return Err(VcsError::Git("malformed `git ls-remote` tag output".to_string()));
        };
        let object = CommitSha::parse(object)
            .map_err(|_| VcsError::Git("`git ls-remote` returned a non-SHA-1 tag object identifier".to_string()))?;
        match reference {
            reference if reference == tag_ref => {
                if tag_object.replace(object).is_some() {
                    return Err(VcsError::Git("`git ls-remote` returned duplicate tag refs".to_string()));
                }
            }
            reference if reference == peeled_ref => {
                if peeled_target.replace(object).is_some() {
                    return Err(VcsError::Git(
                        "`git ls-remote` returned duplicate peeled tag refs".to_string(),
                    ));
                }
            }
            _ => {
                return Err(VcsError::Git(
                    "`git ls-remote` returned an unexpected tag ref".to_string(),
                ))
            }
        }
    }
    Ok(LsRemoteTagRefs {
        tag_object,
        peeled_target,
    })
}

/// SHA-256 of the canonical message bytes in an annotated Git tag.
///
/// `git tag -m` stores a final newline after its message. Release callers
/// construct this value from the logical annotation they pass to
/// [`GitDataSource::create_tag`], while remote observation hashes the raw
/// message bytes in the fetched tag object. Keeping that conversion here
/// prevents the graph layer from having a second, subtly different notion of
/// annotated-tag equality.
#[derive(Clone, Debug, PartialEq, Eq, Hash)]
pub struct TagAnnotationDigest(String);

impl TagAnnotationDigest {
    /// Hashes an annotation as Git stores the ordinary `git tag -m` message.
    pub fn from_annotation(annotation: &str) -> Self {
        let mut bytes = annotation.trim_end_matches(['\r', '\n']).as_bytes().to_vec();
        bytes.push(b'\n');
        Self(format!("{:x}", Sha256::digest(bytes)))
    }

    /// Hashes raw annotation bytes from a parsed annotated tag object.
    pub fn from_tag_object_message(message: &[u8]) -> Self {
        Self(format!("{:x}", Sha256::digest(message)))
    }

    /// Returns the canonical lowercase SHA-256 rendering.
    pub fn as_str(&self) -> &str {
        &self.0
    }
}

impl fmt::Display for TagAnnotationDigest {
    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
        formatter.write_str(&self.0)
    }
}
