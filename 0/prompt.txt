/// A canonical product name used in product release tags.
///
/// Product names are deliberately narrower than package names. They become a
/// git tag component and therefore use lowercase ASCII letters, digits, and
/// interior hyphens only.
#[derive(Clone, Debug, PartialEq, Eq, Hash, PartialOrd, Ord, JsonSchema)]
#[schemars(with = "String")]
pub struct ProductName(String);

impl ProductName {
    /// Parses one canonical product name.
    pub fn parse(raw: &str) -> Result<Self, ProductNameError> {
        if raw.is_empty() {
            return Err(ProductNameError::Empty);
        }
        if raw.starts_with('-') || raw.ends_with('-') {
            return Err(ProductNameError::BoundaryHyphen { raw: raw.to_string() });
        }
        if !raw
            .bytes()
            .all(|byte| byte.is_ascii_lowercase() || byte.is_ascii_digit() || byte == b'-')
        {
            return Err(ProductNameError::InvalidCharacter { raw: raw.to_string() });
        }
        Ok(Self(raw.to_string()))
    }

    /// Returns the canonical name.
    pub fn as_str(&self) -> &str {
        &self.0
    }
}

impl fmt::Display for ProductName {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        f.write_str(&self.0)
    }
}

impl FromStr for ProductName {
    type Err = ProductNameError;

    fn from_str(raw: &str) -> Result<Self, Self::Err> {
        Self::parse(raw)
    }
}
