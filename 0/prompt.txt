    fn observe_remote_annotated_tag(
        &self,
        remote: &str,
        name: &TagName,
        target_sha: &CommitSha,
        annotation_digest: &TagAnnotationDigest,
    ) -> Result<RemoteAnnotatedTagObservation, VcsError> {
        // Ask for both the tag object and its peeled target. `ls-remote`
        // exposes neither a tag annotation nor its object body, so after this
        // cheap absence/target check we fetch only the named tag object with
        // no local ref or FETCH_HEAD update and inspect that exact object.
        let tag_ref = format!("refs/tags/{}", name.as_str());
        let peeled_ref = format!("{tag_ref}^{{}}");
        let ls_remote = self.runner.run(
            "git",
            &["ls-remote", "--tags", "--", remote, &tag_ref, &peeled_ref],
            &self.root,
        )?;
        if !ls_remote.success() {
            return Err(remote_command_error("git ls-remote", &ls_remote));
        }
        let refs = parse_ls_remote_tag_refs(&ls_remote.stdout, &tag_ref, &peeled_ref)?;
        let Some(tag_object) = refs.tag_object else {
            return Ok(RemoteAnnotatedTagObservation::Absent);
        };
        let Some(peeled_target) = refs.peeled_target else {
            return Ok(RemoteAnnotatedTagObservation::Conflict(
                RemoteAnnotatedTagConflict::LightweightTag,
            ));
        };
        if peeled_target != *target_sha {
            return Ok(RemoteAnnotatedTagObservation::Conflict(
                RemoteAnnotatedTagConflict::Target {
                    expected: target_sha.clone(),
                    actual: peeled_target,
                },
            ));
        }

        let fetch = self.runner.run(
            "git",
            &["fetch", "--no-tags", "--no-write-fetch-head", "--", remote, &tag_ref],
            &self.root,
        )?;
        if !fetch.success() {
            return Err(remote_command_error("git fetch", &fetch));
        }
        let tag_object_id = tag_object.as_str();
        let object = self.runner.run(
            "git",
            &["--no-replace-objects", "cat-file", "-p", "--", tag_object_id],
            &self.root,
        )?;
        if !object.success() {
            return Err(remote_command_error("git cat-file", &object));
        }
        let (declared_target, actual_annotation) = parse_annotated_tag_object(&object.stdout)?;
        if declared_target != *target_sha {
            return Ok(RemoteAnnotatedTagObservation::Conflict(
                RemoteAnnotatedTagConflict::MalformedTarget,
            ));
        }
        let actual_digest = TagAnnotationDigest::from_tag_object_message(actual_annotation.as_bytes());
        if actual_digest != *annotation_digest {
            return Ok(RemoteAnnotatedTagObservation::Conflict(
                RemoteAnnotatedTagConflict::Annotation {
                    expected: annotation_digest.clone(),
                    actual: actual_digest,
                },
            ));
        }
        Ok(RemoteAnnotatedTagObservation::Exact)
    }
