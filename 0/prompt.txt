        &self,
        program: &str,
        args: &[&str],
        cwd: &Path,
    ) -> Result<callisto_model::CommandOutput, callisto_model::CommandError> {
        self.0.lock().unwrap().push((

            args.iter().map(|arg| (*arg).to_owned()).collect(),

        ));

            exit_code: Some(0),
            stdout: String::new(),
            stderr: String::new(),

    }
